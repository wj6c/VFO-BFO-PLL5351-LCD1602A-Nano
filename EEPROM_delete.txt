#include <EEPROM.h>

void setup() {
  Serial.begin(38400);
  Serial.println("Borrando EEPROM del VFO...");
  for (int i = 0; i < EEPROM.length(); i++) {
    EEPROM.update(i, 0xFF);
  }
  Serial.println("EEPROM borrada. Sube el VFO normal ahora.");
}
